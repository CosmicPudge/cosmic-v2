===== Widget.tsx =====
"use client";

import { motion } from "framer-motion";
import clsx from "clsx";

import GlassPanel from "@/components/os/ui/GlassPanel";

interface WidgetProps {
  title?: string;
  subtitle?: string;
  children: React.ReactNode;
  className?: string;

  accent?:
    | "default"
    | "weather"
    | "calendar"
    | "sports"
    | "garage"
    | "school"
    | "assistant";

  hover?: boolean;
}

const ACCENTS = {
  default: "from-white/8",
  weather: "from-sky-400/12",
  calendar: "from-violet-400/12",
  sports: "from-green-400/12",
  garage: "from-orange-400/12",
  school: "from-blue-400/12",
  assistant: "from-cyan-400/12",
};

export default function Widget({
  children,
  className,
  hover = true,
  accent = "default",
}: WidgetProps) {
  return (
    <motion.div
      whileHover={
        hover
          ? {
              y: -4,
            }
          : undefined
      }
      transition={{
        type: "spring",
        stiffness: 260,
        damping: 24,
      }}
      className={clsx("h-full", className)}
    >
      <GlassPanel
        hover={hover}
        className="
          relative
          h-full
          overflow-hidden
          rounded-[36px]
          p-8
        "
      >
        {/* Ambient Accent Glow */}
        <div
          className={clsx(
            "pointer-events-none absolute inset-0",
            "bg-[radial-gradient(circle_at_top_left,rgba(255,255,255,.12),transparent_60%)]"
          )}
        />

        {/* Accent Tint */}
        <div
          className={clsx(
            "pointer-events-none absolute inset-0 bg-gradient-to-br to-transparent",
            ACCENTS[accent]
          )}
        />

        {/* Soft Top Highlight */}
        <div
          className="
            pointer-events-none
            absolute
            inset-0
            bg-gradient-to-b
            from-white/[0.06]
            via-transparent
            to-transparent
          "
        />

        <div className="relative z-10 flex h-full flex-col">
          {children}
        </div>
      </GlassPanel>
    </motion.div>
  );
}
===== WidgetHeader.tsx =====
"use client";

interface Props {
  title: string;
  subtitle?: string;
  icon?: React.ReactNode;
  action?: React.ReactNode;
}

export default function WidgetHeader({
  title,
  subtitle,
  icon,
  action,
}: Props) {
  return (
    <header className="mb-8 flex items-start justify-between gap-6">
      <div className="min-w-0 flex-1">
        <div className="flex items-center gap-3">
          {icon && (
            <div className="flex h-9 w-9 items-center justify-center rounded-2xl bg-white/6 text-white/85 backdrop-blur-xl">
              {icon}
            </div>
          )}

          <div className="min-w-0">
            <h3
              className="
                text-[11px]
                font-semibold
                uppercase
                tracking-[0.30em]
                text-white/60
              "
            >
              {title}
            </h3>

            {subtitle && (
              <p
                className="
                  mt-2
                  truncate
                  text-base
                  font-medium
                  text-white/85
                "
              >
                {subtitle}
              </p>
            )}
          </div>
        </div>
      </div>

      {action && (
        <div className="flex shrink-0 items-center">
          {action}
        </div>
      )}
    </header>
  );
}
